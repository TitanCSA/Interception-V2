cmake_minimum_required(VERSION 3.24)
project(InterceptionV2 VERSION 2.0.0 LANGUAGES C CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(interception SHARED library/interception.c library/interception.rc)
target_compile_definitions(interception PRIVATE INTERCEPTION_EXPORT _CRT_SECURE_NO_WARNINGS)
target_include_directories(interception PUBLIC library)
set_target_properties(interception PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/library" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/library")
foreach(name axes cadstop caps2esc hardwareid identify mathpointer x2y)
  add_executable(${name} samples/${name}/${name}.cpp samples/utils.c)
  target_include_directories(${name} PRIVATE library samples)
  target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(${name} PRIVATE interception)
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/samples")
  add_custom_command(TARGET ${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:interception> $<TARGET_FILE_DIR:${name}>)
endforeach()
